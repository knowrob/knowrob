39 bool PredicateNegationStage::succeeds(const AnswerYesPtr &answer) {
40  // here the idea is to issue individual queries for each negative literal.
41  // each query produces an AnswerBufferPtr object.
42  // the set of all results can then be checked for a certain property, e.g. that none or all
43  // of the queries produced a result.
44  std::vector<TokenBufferPtr> results;
45 
46  auto kg = kb_->getBackendForQuery();
47 
48  for (auto &pat: negatedLiterals_) {
49  auto pat1 = applyBindings(pat, *answer->substitution());
50  auto indicator = RDFIndicator(pat1->predicate());
51 
52  // create an instance of the pattern based on given substitution
53  auto instance = std::make_shared<FirstOrderLiteral>(
54  pat1->predicate(), pat1->isNegated());
55  // for now evaluate positive variant of the pattern.
56  // NOTE: for open-world semantics this cannot be done. open-world reasoner
57  // would need to receive negative literal instead.
58  instance->setIsNegated(false);
59 
60  // check if the EDB contains positive lit, if so negation cannot be true
61  if (indicator.arity <= 2) {
62  auto rdfLiteral = std::make_shared<TriplePattern>(
63  instance->predicate(), instance->isNegated());
64  rdfLiteral->setTripleFrame(ctx_->selector);
65  results.push_back(kb_->edb()->getAnswerCursor(
66  kg, std::make_shared<GraphPathQuery>(rdfLiteral, ctx_)));
67  }
68 
69  // next check if positive lit is an IDB predicate, if so negation cannot be true.
70  // get list of reasoner that define the literal
71  if (indicator.functor) {
72  auto l_reasoner = kb_->reasonerManager()->findDefiningReasoner(
73  PredicateIndicator(*indicator.functor, indicator.arity));
74  for (auto &r: l_reasoner) {
75  results.push_back(ReasonerManager::evaluateQuery(r.first, {instance}, ctx_));
76  }
77  }
78  }
79 
80  // go through results, ensure *all* sub-queries failed, only then the negation stage succeeds.
81  for (auto &result: results) {
82  auto resultQueue = result->createQueue();
83  auto firstResult = resultQueue->pop_front();
84  if (firstResult->tokenType() == TokenType::ANSWER_TOKEN) {
85  if (std::static_pointer_cast<const Answer>(firstResult)->isPositive()) {
86  return false;
87  }
88  }
89  }
90 
91  return true;
92 }

100 bool ModalNegationStage::succeeds(const AnswerYesPtr &answer) {
101  // here the idea is to issue individual queries for each negated modal.
102  // for now the positive variants of the modals is evaluated, the modal negation is thought to be
103  // true if none of the positive variants can be grounded.
104 
105  auto outputStream = std::make_shared<TokenBuffer>();
106 
107  // run "submitQuery" for each negated modal, and collect results in outputStream
108  for (auto &modal: negatedModals_) {
109  auto modalInstance = applyBindings(modal, *answer->substitution());
110  auto modalOutput = kb_->submitQuery(modalInstance, ctx_);
111  modalOutput >> outputStream;
112  modalOutput->stopBuffering();
113  }
114 
115  // wait for and pop first message from output stream
116  auto resultQueue = outputStream->createQueue();
117  auto firstResult = resultQueue->pop_front();
118  // "EOS" indicates that no solution was found for any of the modals
119  return firstResult->indicatesEndOfEvaluation();
120 }

14 ModalStage::ModalStage(
15  const std::shared_ptr<KnowledgeBase> &kb,
16  const std::shared_ptr<ModalFormula> &modal,
17  const QueryContextPtr &ctx)
18  : TypedQueryStage(ctx, modal->modalFormula(),
19  [this](auto &&PH1) { return submitQuery(std::forward<decltype(PH1)>(PH1)); }),
20  kb_(kb),
21  modalFormula_(modal) {
22  // this modal stage could be embedded into another modal expression.
23  // however, forming a mental model of what a human might think about what the robot thinks about him is an
24  // additional level which we do not consider (yet).
25  // However, expressions like `P(Bp & Bq)` with iterations over different modality types can be used in queries.
26 
27  auto modalContext = std::make_shared<QueryContext>(
28  *ctx, modalFormula_->modalOperator());
29  nestedContext_ = modalContext;
30 
31  auto &modalOperator = modalFormula_->modalOperator();
32 
33  // modify the query context for the evaluation of the modal.
34  switch (modalOperator->modalType()) {
35  case ModalType::KNOWLEDGE:
36  case ModalType::BELIEF:
37  // epistemic states of different agents are stored in a single KG where triples are labeled by the agent.
38  // These states are assumed to be generated from the perspective of a "self", i.e. the agent
39  // that runs the knowledge base and its information about the other agents.
40  if (ctx->selector.perspective && ctx->selector.perspective != Perspective::getEgoPerspective()) {
41  // For now higher-level epistemic states are not allowed in queries,
42  // e.g. `B_a1(B_a2(x))` where `a1` is not the agent that runs the knowledge base is an example
43  // of such a higher-order query which is not allowed.
44  // `B_self(B_a2(x))` is ok though, and changes the agent label of the KG selector to `a2` below.
45  throw QueryError("epistemic formula {} is embedded within epistemic context of another agent.",
46  *modalFormula_);
47  }
48  if (modalOperator->perspective()) {
49  modalContext->selector.perspective = *modalOperator->perspective();
50  } else {
51  modalContext->selector.perspective = nullptr;
52  }
53  modalContext->selector.uncertain = !modalOperator->isModalNecessity();
54  if (modalContext->selector.uncertain) {
55  modalContext->selector.confidence = modalOperator->confidence();
56  }
57  break;
58 
59  case ModalType::ALWAYS:
60  case ModalType::SOMETIMES:
61  modalContext->selector.occasional = !modalOperator->isModalNecessity();
62  modalContext->selector.begin = modalOperator->begin();
63  modalContext->selector.end = modalOperator->end();
64  break;
65  }
66 }

68 TokenBufferPtr ModalStage::submitQuery_K(const FormulaPtr &phi) {
69  // This case corresponds to "I know that `phi` is true".
70  // It is assumed that all the knowledge of an agent is stored in an EDB or can be derived
71  // from EDB-facts through top-down methods.
72  // Hence, `phi` can be evaluated with the constraint that all positive literals can be grounded
73  // in or derived from EDB-facts that are marked as "certain".
74  // NOTE: At a later point it might be interesting to connect this evaluation to a simulation model
75  // where the world is simulated from the perspective of another agent.
76  return kb_->submitQuery(phi, nestedContext_);
77 }

79 TokenBufferPtr ModalStage::submitQuery_B(const FormulaPtr &phi) {
80  // This case corresponds to "I believe that `phi` is true".
81  // This is interpreted as a "best guess", and it is assumed that there is only one
82  // best guess at a time which is materialized in the EDB.
83  // Hence, `phi` can be evaluated with the constraint that all positive literals can be grounded
84  // in or derived from EDB-facts that are marked as "certain" or "uncertain".
85  return kb_->submitQuery(phi, nestedContext_);

88 TokenBufferPtr ModalStage::submitQuery_P(const FormulaPtr &phi) {
89  // This case corresponds to "At some point in the past `phi` was true".
90  // It is assumed here that the past is materialized in the EDB already.
91  // One use case being the NEEMs of the EASE project that represent episodes
92  // of robot actions that happened in the past.
93  // Hence, `phi` can be evaluated with the constraint that all positive literals can be grounded
94  // in or derived from EDB-facts whose temporal intersection is non-empty:
95  // NOTE: At a later point it might be interesting to consider this case and below as
96  // access to long term memory where the KG is not materialized locally.
97  // e.g. using the NEEMHub infrastructure of EASE directly for query evaluation.
98  return kb_->submitQuery(phi, nestedContext_);
99 }

109 TokenBufferPtr ModalStage::submitQuery(const FormulaPtr &phi) {
110  switch (modalFormula_->modalOperator()->modalType()) {
111  case ModalType::KNOWLEDGE:
112  return submitQuery_K(phi);
113  case ModalType::BELIEF:
114  return submitQuery_B(phi);
115  case ModalType::ALWAYS:
116  return submitQuery_H(phi);
117  case ModalType::SOMETIMES:
118  return submitQuery_P(phi);
119  }
120  throw QueryError("unexpected modality type in query '{}'.", *modalFormula_);
121 }
